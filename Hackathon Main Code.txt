    const ctx = getAudioContext();
    const bufferSize = ctx.sampleRate * 2;
    const buffer = ctx.createBuffer(1, bufferSize, ctx.sampleRate);
    const data = buffer.getChannelData(0);
    let lastOut = 0;
    for (let i = 0; i < bufferSize; i++) {
      if (type === 'pink') {
        const white = Math.random() * 2 - 1;
        lastOut += (white - lastOut) * 0.02;
        data[i] = lastOut * 0.5;
      } else if (type === 'brown') {
        const white = Math.random() * 2 - 1;
        lastOut += (white - lastOut) * 0.1;
        if (lastOut > 1) lastOut = 1;
        if (lastOut < -1) lastOut = -1;
        data[i] = lastOut * 0.4;
      } else {
        data[i] = Math.random() * 2 - 1;
      }
    }
    return buffer;

  window.generateTimetable = function() {
    if (state.subjects.length === 0) {
      showToast('Add at least one subject first');
      return;
    }

    const prefs = getPreferences();
    const mode = state.selectedMode || prefs.intensity || 'balanced';

    let numSessions, focusDuration, breakDuration;
    if (mode === 'chill') {
      numSessions = 3; focusDuration = 25; breakDuration = 12;
    } else if (mode === 'grind') {
      numSessions = 6; focusDuration = 42; breakDuration = 12;
    } else {
      numSessions = 4; focusDuration = 28; breakDuration = 10;
    }

    // Sort subjects by urgency (low confidence + approaching exam = high priority)
    const sorted = [...state.subjects].sort((a, b) => {
      const daysUntilA = (new Date(a.exam) - new Date()) / (1000*60*60*24);
      const daysUntilB = (new Date(b.exam) - new Date()) / (1000*60*60*24);
      const urgencyA = (11 - a.confidence) * (1 / Math.max(daysUntilA, 1));
      const urgencyB = (11 - b.confidence) * (1 / Math.max(daysUntilB, 1));
      return urgencyB - urgencyA;
    });

    const wakeParts = prefs.wakeTime.split(':').map(Number);
    const wakeMin = wakeParts[0] * 60 + wakeParts[1];
    const availMin = prefs.availableHours * 60;
    const totalSessionTime = numSessions * focusDuration;
    const totalBreakTime = (numSessions - 1) * prefs.breakDuration + (numSessions >= 3 ? 40 : 20); // meal buffers
    const neededTime = totalSessionTime + totalBreakTime;
    const startMin = wakeMin + 30; // start 30min after waking

    const bedtimeMin = calculateBedtime(wakeMin, prefs.sleepDuration);
    const stopStudyMin = bedtimeMin - prefs.stopBuffer;
    const maxEnd = Math.min(stopStudyMin, startMin + availMin);

    const timetable = [];
    let currentMin = startMin;
    let sessionCount = 0;
    let prevSubjectId = null;

    const methods = ['Active Recall', 'Practice Questions', 'Interleaving', 'Spaced Repetition', 'Self Test', 'Light Review'];

    for (let i = 0; i < numSessions; i++) {
      if (currentMin + focusDuration > maxEnd) break;

      // Pick subject
      let subject;
      if (i === 0) {
        subject = sorted[0]; // hardest/most urgent first
      } else if (i === 2 && sorted.length > 1) {
        subject = sorted[1]; // different subject for interleaving
      } else if (i === 3 && sorted.length > 2) {
        subject = sorted[2];
      } else {
        // alternate subjects
        const candidates = sorted.filter(s => s.id !== prevSubjectId);
        subject = candidates.length > 0 ? candidates[i % candidates.length] : sorted[i % sorted.length];
      }
      if (!subject) subject = sorted[i % sorted.length];

      // Pick topic
      let topic;
      const weakTopics = subject.weakTopics || [];
      const allTopics = subject.topics || [];
      if (i === 0 && weakTopics.length > 0) {
        topic = weakTopics[0]; // weakest topic first
      } else if (allTopics.length > 0) {
        topic = allTopics[i % allTopics.length];
      } else {
        topic = 'Main topics';
      }

      // Pick method
      const method = i < methods.length ? methods[i] : methods[i % methods.length];

      const methodInstructions = {
        'Active Recall': 'Close your notes and write down everything you remember before checking.',
        'Practice Questions': 'Attempt questions before reading the solutions.',
        'Interleaving': 'Switch subjects to improve long-term understanding.',
        'Spaced Repetition': 'Review information before forgetting begins.',
        'Self Test': 'Simulate exam conditions and answer without help.',
        'Light Review': 'Summarize key ideas without introducing difficult new concepts.'
      };

      const session = {
        id: state.nextId++,
        type: 'study',
        subjectId: subject.id,
        subjectName: subject.name,
        topic: topic,
        method: method,
        instruction: methodInstructions[method] || 'Focus on understanding the material.',
        duration: focusDuration,
        startMin: currentMin,
        endMin: currentMin + focusDuration,
        completed: false,
        skipped: false,
        status: 'pending',
        reasoning: ''
      };
      session.reasoning = generateReason(session, subject);
      timetable.push(session);

      prevSubjectId = subject.id;
      currentMin += focusDuration;
      sessionCount++;

      // Add break after each session except last
      if (i < numSessions - 1) {
        let breakType = 'short';
        let breakLen = prefs.breakDuration || breakDuration;

        // Insert meal breaks after session 2 and 4
        if (sessionCount === 2) {
          breakType = 'lunch';
          breakLen = 25;
        } else if (sessionCount === 4) {
          breakType = 'dinner';
          breakLen = 25;
        }

        timetable.push({
          id: state.nextId++,
          type: breakType === 'lunch' || breakType === 'dinner' ? 'meal' : 'break',
          breakType: breakType,
          duration: breakLen,
          startMin: currentMin,
          endMin: currentMin + breakLen,
          completed: false,
          skipped: false,
          status: 'pending',
          label: breakType === 'lunch' ? '🍝 Lunch Break' : breakType === 'dinner' ? '🍽️ Dinner Break' : '☕ Short Break'
        });
        currentMin += breakLen;
      }
    }

    // Check if end is too close to bedtime
    if (currentMin + prefs.windDown > stopStudyMin && prefs.protectSleep) {
      // Auto-shift sessions earlier if possible
      const overrun = (currentMin + prefs.windDown) - stopStudyMin;
      if (overrun > 15) {
        showToast('Sessions adjusted to protect your sleep. Review the schedule.');
      }
    }

    // Add wind-down and bedtime as informational blocks
    const windDownStart = Math.max(currentMin, stopStudyMin - prefs.windDown);
    timetable.push({
      id: state.nextId++,
      type: 'winddown',
      duration: prefs.windDown,
      startMin: windDownStart,
      endMin: windDownStart + prefs.windDown,
      completed: false,
      skipped: false,
      status: 'pending',
      label: '🌙 Wind Down'
    });

    const bedMin = windDownStart + prefs.windDown;
    timetable.push({
      id: state.nextId++,
      type: 'bedtime',
      duration: prefs.sleepDuration * 60,
      startMin: bedMin,
      endMin: bedMin + prefs.sleepDuration * 60,
      completed: false,
      skipped: false,
      status: 'pending',
      label: '😴 Bedtime'
    });

    state.timetable = timetable;
    localStorage.setItem('studentos_timetable', JSON.stringify(timetable));

    // Update streak
    const today = new Date().toDateString();
    const lastActive = localStorage.getItem('studentos_last_active');
    if (lastActive !== today) {
      state.streak++;
      localStorage.setItem('studentos_streak', String(state.streak));
      localStorage.setItem('studentos_last_active', today);
    }

    showToast('Study plan generated! 🎉');
    renderAllPages();
    navigateTo('timetable');
  };

  function calculateBedtime(wakeMin, sleepHours) {
    let bedMin = wakeMin - sleepHours * 60;
    if (bedMin < 0) bedMin += 24 * 60;
    return bedMin;
  }

  function formatTime(minutes) {
    const h = Math.floor(minutes / 60) % 24;
    const m = minutes % 60;
    return `${String(h).padStart(2, '0')}:${String(m).padStart(2, '0')}`;
  }

  function formatAmPm(minutes) {
    const h = Math.floor(minutes / 60) % 24;
    const m = minutes % 60;
    const period = h >= 12 ? 'PM' : 'AM';
    const h12 = h % 12 || 12;
    return `${h12}:${String(m).padStart(2, '0')} ${period}`;
  }

  window.skipSession = function(id) {
    rescheduleTargetId = id;
    const session = state.timetable.find(s => s.id === id);
    if (!session) return;
    document.getElementById('rescheduleInfo').textContent =

  window.rescheduleAction = function(action) {
    const idx = state.timetable.findIndex(s => s.id === rescheduleTargetId);
    if (idx === -1) { closeModal('rescheduleModal'); return; }

    const session = state.timetable[idx];
    if (action === 'remove') {
      state.timetable.splice(idx, 1);
      showToast('Session removed');
    } else if (action === 'next') {
      // Move to next available slot (after last session, before wind-down)
      const lastSession = [...state.timetable].reverse().find(s => s.type === 'study');
      if (lastSession && session.type === 'study') {
        const gap = lastSession.endMin + 15;
        const winddown = state.timetable.find(s => s.type === 'winddown');
        if (winddown && gap + session.duration < winddown.startMin) {
          session.startMin = gap;
          session.endMin = gap + session.duration;
          session.skipped = false;
          showToast('Moved to next slot');
        } else {
          showToast('No available slot before wind-down. Try tomorrow.');
        }
      }
    } else if (action === 'tomorrow') {
      // Keep in timetable but mark for tomorrow
      session.rescheduledTomorrow = true;
      session.skipped = false;
      showToast('Moved to tomorrow');
    }

    localStorage.setItem('studentos_timetable', JSON.stringify(state.timetable));
    closeModal('rescheduleModal');
    renderTimetable();
  };

  function updateFocusTimer() {

      `${String(mins).padStart(2, '0')}:${String(secs).padStart(2, '0')}`;

    const circumference = 2 * Math.PI * 70;
    const offset = circumference * (1 - state.focusRemaining / state.focusTotal);
    document.getElementById('focusProgressRing').style.strokeDashoffset = offset;
  }

    const idx = state.focusSessionIndex;

  // ---- SLEEP COACH ----
  function calculateSleep(prefs) {
    const wakeParts = prefs.wakeTime.split(':').map(Number);
    const wakeMin = wakeParts[0] * 60 + wakeParts[1];
    const sleepMin = prefs.sleepDuration * 60;
    let bedMin = wakeMin - sleepMin;
    if (bedMin < 0) bedMin += 24 * 60;
    const windDownMin = bedMin - prefs.windDown;
    let stopStudyMin = bedMin - prefs.stopBuffer;
    if (stopStudyMin < 0) stopStudyMin += 24 * 60;

    return {
      bedtime: formatTime(bedMin),
      bedtimeAmPm: formatAmPm(bedMin),
      windDown: formatTime(windDownMin),
      windDownAmPm: formatAmPm(windDownMin),
      stopStudying: formatTime(stopStudyMin),
      stopStudyingAmPm: formatAmPm(stopStudyMin),
      expectedSleep: prefs.sleepDuration,
      wakeTime: prefs.wakeTime,
      isWarning: prefs.sleepDuration < 7
    };
  }

  // ---- ICS EXPORT ----
  window.exportICS = function() {
    const prefs = getPreferences();
    const sleep = calculateSleep(prefs);
    let ics = [
      'BEGIN:VCALENDAR',
      'VERSION:2.0',
      'PRODID:-//StudentOS//AI Planner//EN',
      'CALSCALE:GREGORIAN'
    ];

    const today = new Date();
    const dateStr = today.toISOString().replace(/[-:]/g, '').split('.')[0] + 'Z';

    state.timetable.forEach(session => {
      const startDate = new Date(today);
      startDate.setHours(Math.floor(session.startMin / 60), session.startMin % 60, 0);
      const endDate = new Date(today);
      endDate.setHours(Math.floor(session.endMin / 60), session.endMin % 60, 0);

      const dtStart = startDate.toISOString().replace(/[-:]/g, '').split('.')[0] + 'Z';
      const dtEnd = endDate.toISOString().replace(/[-:]/g, '').split('.')[0] + 'Z';

      let summary = '';
      if (session.type === 'study') {
        summary = `Study: ${session.subjectName} - ${session.topic} (${session.method})`;
      } else if (session.type === 'break') {
        summary = session.label || 'Break';
      } else if (session.type === 'meal') {
        summary = session.label || 'Meal Break';
      } else if (session.type === 'winddown') {
        summary = '🌙 Wind Down';
      } else if (session.type === 'bedtime') {
        summary = '😴 Sleep';
      }

      ics.push('BEGIN:VEVENT');
      ics.push(`DTSTART:${dtStart}`);
      ics.push(`DTEND:${dtEnd}`);
      ics.push(`SUMMARY:${summary}`);
      ics.push('END:VEVENT');
    });

    // Add wind-down and bedtime
    if (prefs.calendarExport) {
      const wdDate = new Date(today);
      const wdParts = sleep.windDown.split(':').map(Number);
      wdDate.setHours(wdParts[0], wdParts[1], 0);
      const wdEnd = new Date(wdDate);
      wdEnd.setMinutes(wdEnd.getMinutes() + prefs.windDown);
      ics.push('BEGIN:VEVENT');
      ics.push(`DTSTART:${wdDate.toISOString().replace(/[-:]/g, '').split('.')[0]}Z`);
      ics.push(`DTEND:${wdEnd.toISOString().replace(/[-:]/g, '').split('.')[0]}Z`);
      ics.push('SUMMARY:🌙 Wind Down');
      ics.push('DESCRIPTION:Begin winding down for sleep');
      ics.push('END:VEVENT');

      const bedDate = new Date(today);
      const bedParts = sleep.bedtime.split(':').map(Number);
      bedDate.setHours(bedParts[0], bedParts[1], 0);
      const bedEnd = new Date(bedDate);
      bedEnd.setHours(bedEnd.getHours() + prefs.sleepDuration);
      ics.push('BEGIN:VEVENT');
      ics.push(`DTSTART:${bedDate.toISOString().replace(/[-:]/g, '').split('.')[0]}Z`);
      ics.push(`DTEND:${bedEnd.toISOString().replace(/[-:]/g, '').split('.')[0]}Z`);
      ics.push('SUMMARY:😴 Sleep');
      ics.push('END:VEVENT');
    }

    ics.push('END:VCALENDAR');

    const blob = new Blob([ics.join('\r\n')], { type: 'text/calendar;charset=utf-8' });
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = `studentos-schedule-${today.toISOString().split('T')[0]}.ics`;
    a.click();
    URL.revokeObjectURL(url);
    showToast('Calendar exported! 📅');
  };

  window.toggleSound = function(id, type, label) {
    if (soundStates[id] && soundStates[id].playing) {
      stopSound(id);
    } else {
      // Stop other sounds in same category
      const isFocus = ['piano','lofi','ambient','deepfocus'].includes(type);
      Object.keys(soundStates).forEach(key => {
        const s = soundStates[key];
        if (s && s.playing) {
          const sIsFocus = ['piano','lofi','ambient','deepfocus'].includes(s.type);
          if (isFocus === sIsFocus || key !== id) stopSound(key);
        }
      });
      startSound(id, type, label);
    }
  };
